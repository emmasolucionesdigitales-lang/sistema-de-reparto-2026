      var p = '='.repeat((4 - b.length % 4) % 4);
      var s = (b + p).replace(/-/g,'+').replace(/_/g,'/');
      var r = atob(s); var o = new Uint8Array(r.length);

    }
    function idActual(){ return window._negocioId || 'firebase'; }

    async function guardarSub(sub){
      if(!window.db) return;
      var id = idActual();
      var deviceId = deviceIdActual();
      // Se guarda bajo la clave de ESTE dispositivo — no pisa la suscripción
      // de otro celu/PC que use el mismo negocio.
      var payload = {};
      payload[deviceId] = { sub: JSON.stringify(sub), ts: Date.now() };
      await window.db.collection('users').doc(id).collection('datos').doc('push_subs').set(payload, { merge:true });
    }
    async function activarNotif(){
      var p = await Notification.requestPermission();

      var reg = await navigator.serviceWorker.ready;
      var subVieja = await reg.pushManager.getSubscription();
      if(subVieja){ try{ await subVieja.unsubscribe(); }catch(e){} } // siempre limpio, evita quedar con endpoint viejo

      return true;
